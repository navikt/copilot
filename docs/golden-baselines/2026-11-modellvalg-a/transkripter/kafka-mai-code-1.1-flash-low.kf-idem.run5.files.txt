package no.nav.vedtak

import org.apache.kafka.clients.consumer.Consumer
import java.time.Duration

fun interface Utbetaling {
    fun utbetal(eventId: String, belop: Int)
}

// Leser vedtak fra topicen og utbetaler. Verdien i hver melding er "eventId;belop".
class VedtakKonsument(
    private val consumer: Consumer<String, String>,
    private val utbetaling: Utbetaling,
) {
    private val utbetalteEventIder: MutableSet<String> = mutableSetOf()

    constructor(
        consumer: Consumer<String, String>,
        utbetaling: Utbetaling,
        utbetalteEventIder: MutableSet<String>,
    ) : this(consumer, utbetaling) {
        this.utbetalteEventIder.addAll(utbetalteEventIder)
    }

    fun poll() {
        val records = consumer.poll(Duration.ofMillis(100))

        for (record in records) {
            val (eventId, belop) = record.value().split(";", limit = 2)
            if (utbetalteEventIder.add(eventId)) {
                utbetaling.utbetal(eventId, belop.toInt())
            }
        }

        if (records.count() > 0) {
            consumer.commitSync()
        }
    }
}

package no.nav.vedtak

import org.apache.kafka.clients.consumer.Consumer
import org.apache.kafka.clients.consumer.ConsumerRecord
import org.apache.kafka.clients.consumer.MockConsumer
import org.apache.kafka.clients.consumer.OffsetResetStrategy
import org.apache.kafka.common.TopicPartition
import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class VedtakKonsumentTest {
    private val tp = TopicPartition("vedtak", 0)

    private class TrackingConsumer(
        private val delegate: MockConsumer<String, String>,
    ) : Consumer<String, String> by delegate {
        var commitCalls = 0
            private set

        override fun commitSync() {
            commitCalls++
            delegate.commitSync()
        }
    }

    private fun consumer(vararg verdier: String): TrackingConsumer {
        val mockConsumer = MockConsumer<String, String>(OffsetResetStrategy.EARLIEST).apply {
            assign(listOf(tp))
            updateBeginningOffsets(mapOf(tp to 0L))
            verdier.forEachIndexed { i, v -> addRecord(ConsumerRecord("vedtak", 0, i.toLong(), "sak-1", v)) }
        }
        return TrackingConsumer(mockConsumer)
    }

    @Test
    fun `utbetaler hvert vedtak`() {
        val utbetalt = mutableListOf<String>()
        VedtakKonsument(consumer("e1;100", "e2;200")) { id, _ -> utbetalt += id }.poll()
        assertEquals(listOf("e1", "e2"), utbetalt)
    }

    @Test
    fun `utbetaler samme eventId bare en gang`() {
        val utbetalt = mutableListOf<String>()
        VedtakKonsument(consumer("e1;100", "e1;100", "e2;200")) { id, _ -> utbetalt += id }.poll()
        assertEquals(listOf("e1", "e2"), utbetalt)
    }

    @Test
    fun `committer ikke offset før alle meldinger er behandlet`() {
        val consumer = consumer("e1;100", "e2;200")
        val utbetalt = mutableListOf<String>()
        val konsument = VedtakKonsument(consumer) { id, _ ->
            utbetalt += id
            if (id == "e2") {
                throw RuntimeException("utbetaling feilet")
            }
        }

        val exception = assertFailsWith<RuntimeException> {
            konsument.poll()
        }
        assertEquals("utbetaling feilet", exception.message)
        assertEquals(listOf("e1", "e2"), utbetalt)
        assertEquals(0, consumer.commitCalls)
    }
}
