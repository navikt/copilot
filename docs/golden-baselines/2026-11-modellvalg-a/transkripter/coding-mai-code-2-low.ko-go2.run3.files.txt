package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker.
func Rapport(navn string, grunnlag int, grad ...int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad...))
}

=== ./ytelse/utbetaling.go
package ytelse

// Utbetaling gir dagsatsen: grunnlaget delt på 260 virkedager,
// justert etter ytelsesgraden og rundet ned.
func Utbetaling(grunnlag int, grad ...int) int {
	ytelsesgrad := 100
	if len(grad) > 0 {
		ytelsesgrad = grad[0]
	}

	grunnlagPerProsent := grunnlag / 26000
	rest := grunnlag % 26000
	return grunnlagPerProsent*ytelsesgrad + rest*ytelsesgrad/26000
}
