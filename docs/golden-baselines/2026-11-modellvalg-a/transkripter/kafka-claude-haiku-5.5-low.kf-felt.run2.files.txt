package no.nav.vedtak

import com.fasterxml.jackson.annotation.JsonIgnoreProperties
import com.fasterxml.jackson.module.kotlin.jacksonObjectMapper
import com.fasterxml.jackson.module.kotlin.readValue

enum class Sakstype { ORDINAER, KLAGE, ANKE }

// Hendelsen publiseres på topicen vedtak-fattet og leses av flere team.
// Meldinger uten sakstype leses som ORDINAER. Ukjente felt fra nyere produsenter ignoreres.
@JsonIgnoreProperties(ignoreUnknown = true)
data class VedtakFattet(
    val vedtakId: String,
    val belop: Int,
    val sakstype: Sakstype = Sakstype.ORDINAER,
)

object VedtakSerde {
    private val mapper = jacksonObjectMapper()

    fun les(json: String): VedtakFattet = mapper.readValue(json)

    fun skriv(hendelse: VedtakFattet): String = mapper.writeValueAsString(hendelse)
}

package no.nav.vedtak

import kotlin.test.Test
import kotlin.test.assertEquals

class VedtakFattetTest {
    @Test
    fun `leser en hendelse`() {
        assertEquals(VedtakFattet("v1", 100), VedtakSerde.les("""{"vedtakId":"v1","belop":100}"""))
    }

    @Test
    fun `melding uten sakstype leses som ORDINAER`() {
        assertEquals(Sakstype.ORDINAER, VedtakSerde.les("""{"vedtakId":"v1","belop":100}""").sakstype)
    }

    @Test
    fun `leser sakstype KLAGE og ANKE`() {
        assertEquals(Sakstype.KLAGE, VedtakSerde.les("""{"vedtakId":"v1","belop":100,"sakstype":"KLAGE"}""").sakstype)
        assertEquals(Sakstype.ANKE, VedtakSerde.les("""{"vedtakId":"v1","belop":100,"sakstype":"ANKE"}""").sakstype)
    }

    @Test
    fun `ignorerer ukjente felt fra nyere produsenter`() {
        assertEquals(
            VedtakFattet("v1", 100, Sakstype.KLAGE),
            VedtakSerde.les("""{"vedtakId":"v1","belop":100,"sakstype":"KLAGE","nyttFelt":"x"}"""),
        )
    }

    @Test
    fun `skriver sakstype`() {
        assertEquals(
            VedtakFattet("v1", 100, Sakstype.ANKE),
            VedtakSerde.les(VedtakSerde.skriv(VedtakFattet("v1", 100, Sakstype.ANKE))),
        )
    }
}
