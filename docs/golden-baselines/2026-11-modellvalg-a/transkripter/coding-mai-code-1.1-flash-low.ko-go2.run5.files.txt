=== ./ytelse/rapport.go
package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker for en gitt ytelsesgrad.
func Rapport(navn string, grunnlag int, grad int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad))
}

=== ./ytelse/utbetaling.go
package ytelse

// Utbetaling gir dagsatsen for en gitt grad av full ytelse.
// Graden er prosent av full stilling, der 100 er fullt arbeid og 50 er halv stilling.
func Utbetaling(grunnlag int, grad int) int {
	return (grunnlag * grad) / (260 * 100)
}
