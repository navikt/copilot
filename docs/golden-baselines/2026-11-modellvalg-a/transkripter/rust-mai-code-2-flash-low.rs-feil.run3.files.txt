/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, PartialEq, thiserror::Error)]
pub enum VedtakFeil {
    #[error("Mangler id")]
    ManglerId,
    #[error("Mangler belop")]
    ManglerBelop,
    #[error("Ugyldig belop: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler.next().unwrap_or("").trim();
    if id.is_empty() {
        return Err(VedtakFeil::ManglerId);
    }

    let belop_text = deler.next().ok_or(VedtakFeil::ManglerBelop)?;
    if belop_text.is_empty() {
        return Err(VedtakFeil::ManglerBelop);
    }

    let belop = belop_text
        .parse()
        .map_err(|_| VedtakFeil::UgyldigBelop(belop_text.to_string()))?;

    Ok(Vedtak {
        id: id.to_string(),
        belop,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(
            les("v1;250"),
            Ok(Vedtak {
                id: "v1".into(),
                belop: 250,
            })
        );
    }

    #[test]
    fn mangler_id() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn mangler_belop() {
        assert_eq!(les("v1"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn ugyldig_belop_inneholder_den_ugyldige_teksten() {
        let feil = les("v1;abc").unwrap_err();
        assert_eq!(feil, VedtakFeil::UgyldigBelop("abc".into()));
        assert!(feil.to_string().contains("abc"));
    }
}
