use thiserror::Error;

/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, Error, PartialEq)]
pub enum VedtakFeil {
    #[error("mangler id")]
    ManglerId,
    #[error("mangler belop")]
    ManglerBelop,
    #[error("ugyldig belop: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let (id, belop) = linje
        .split_once(';')
        .ok_or(VedtakFeil::ManglerBelop)?;

    if id.trim().is_empty() {
        return Err(VedtakFeil::ManglerId);
    }
    if belop.is_empty() {
        return Err(VedtakFeil::ManglerBelop);
    }

    let belop = belop
        .parse()
        .map_err(|_| VedtakFeil::UgyldigBelop(belop.to_string()))?;

    Ok(Vedtak {
        id: id.to_string(),
        belop,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(
            les("v1;250"),
            Ok(Vedtak { id: "v1".into(), belop: 250 })
        );
    }

    #[test]
    fn gir_feil_nar_id_mangler() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn gir_feil_nar_belop_mangler() {
        assert_eq!(les("v1"), Err(VedtakFeil::ManglerBelop));
        assert_eq!(les("v1;"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn gir_feil_nar_belop_er_ugyldig() {
        let feil = les("v1;abc").unwrap_err();
        assert_eq!(feil, VedtakFeil::UgyldigBelop("abc".into()));
        assert!(feil.to_string().contains("abc"));
    }
}
