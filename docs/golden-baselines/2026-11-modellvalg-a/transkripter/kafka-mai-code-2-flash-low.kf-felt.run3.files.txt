package no.nav.vedtak

import com.fasterxml.jackson.databind.DeserializationFeature
import com.fasterxml.jackson.module.kotlin.jacksonObjectMapper
import com.fasterxml.jackson.module.kotlin.readValue

enum class Sakstype {
    ORDINAER,
    KLAGE,
    ANKE,
}

// Hendelsen publiseres på topicen vedtak-fattet og leses av flere team.
data class VedtakFattet(
    val vedtakId: String,
    val belop: Int,
    val sakstype: Sakstype = Sakstype.ORDINAER,
)

object VedtakSerde {
    private val mapper = jacksonObjectMapper().apply {
        configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false)
    }

    fun les(json: String): VedtakFattet = mapper.readValue(json)

    fun skriv(hendelse: VedtakFattet): String = mapper.writeValueAsString(hendelse)
}

package no.nav.vedtak

import kotlin.test.Test
import kotlin.test.assertEquals

class VedtakFattetTest {
    @Test
    fun `leser eldre hendelse som ORDINAER`() {
        assertEquals(VedtakFattet("v1", 100), VedtakSerde.les("""{"vedtakId":"v1","belop":100}"""))
    }

    @Test
    fun `leser gyldige sakstyper`() {
        assertEquals(
            VedtakFattet("v1", 100, Sakstype.KLAGE),
            VedtakSerde.les("""{"vedtakId":"v1","belop":100,"sakstype":"KLAGE"}"""),
        )
        assertEquals(
            VedtakFattet("v2", 200, Sakstype.ANKE),
            VedtakSerde.les("""{"vedtakId":"v2","belop":200,"sakstype":"ANKE"}"""),
        )
    }

    @Test
    fun `skriver sakstype`() {
        assertEquals(
            """{"vedtakId":"v1","belop":100,"sakstype":"ORDINAER"}""",
            VedtakSerde.skriv(VedtakFattet("v1", 100)),
        )
    }

    @Test
    fun `tolererer ukjente felter`() {
        assertEquals(
            VedtakFattet("v1", 100),
            VedtakSerde.les(
                """{"vedtakId":"v1","belop":100,"sakstype":"ORDINAER","nyttFelt":{"verdi":"x"}}""",
            ),
        )
    }
}
