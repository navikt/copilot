use thiserror::Error;

/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, Error, PartialEq)]
pub enum VedtakFeil {
    #[error("mangler id")]
    ManglerId,
    #[error("mangler beløp")]
    ManglerBelop,
    #[error("ugyldig beløp: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler.next().unwrap_or_default();
    if id.is_empty() {
        return Err(VedtakFeil::ManglerId);
    }

    let belop_tekst = deler.next();
    let belop_tekst = match belop_tekst {
        Some(tekst) if !tekst.is_empty() => tekst,
        Some(_) => return Err(VedtakFeil::ManglerBelop),
        None => return Err(VedtakFeil::ManglerBelop),
    };

    let belop = belop_tekst.parse::<u32>().map_err(|_| VedtakFeil::UgyldigBelop(belop_tekst.to_string()))?;

    Ok(Vedtak {
        id: id.to_string(),
        belop,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(les("v1;250"), Ok(Vedtak { id: "v1".into(), belop: 250 }));
    }

    #[test]
    fn feiler_naar_id_mangler() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn feiler_naar_belop_mangler() {
        assert_eq!(les("v1"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn feiler_naar_belop_er_ugyldig() {
        let feil = les("v1;abc").unwrap_err();
        assert_eq!(feil, VedtakFeil::UgyldigBelop("abc".into()));
        assert_eq!(feil.to_string(), "ugyldig beløp: abc");
    }
}
