package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker.
func Rapport(navn string, grunnlag, grad int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad))
}

=== ./ytelse/utbetaling.go
package ytelse

// Utbetaling gir dagsatsen: grunnlaget delt på 260 virkedager, ganget med grad (prosent av full ytelse), rundet ned.
// Deling skjer til slutt for å unngå avrundingstap i mellomregningen.
func Utbetaling(grunnlag, grad int) int {
	return grunnlag * grad / (260 * 100)
}
