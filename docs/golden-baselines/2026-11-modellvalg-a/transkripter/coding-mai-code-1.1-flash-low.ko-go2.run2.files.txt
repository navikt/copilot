=== ./ytelse/rapport.go
package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker for en gitt grad.
func Rapport(navn string, grunnlag, grad int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad))
}

=== ./ytelse/utbetaling.go
package ytelse

// Utbetaling gir dagsatsen for en gradert ytelse: full dagsats er grunnlaget
// delt på 260 virkedager, og dette multipliseres med arbeidsgraden i prosent.
func Utbetaling(grunnlag, grad int) int {
	return (grunnlag / 260) * grad / 100
}
