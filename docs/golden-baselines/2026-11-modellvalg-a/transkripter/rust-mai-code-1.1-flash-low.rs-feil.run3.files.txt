use thiserror::Error;

/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, Error, PartialEq)]
pub enum VedtakFeil {
    #[error("mangler id")]
    ManglerId,
    #[error("mangler beløp")]
    ManglerBelop,
    #[error("ugyldig beløp: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler.next().unwrap_or_default().to_string();
    if id.trim().is_empty() {
        return Err(VedtakFeil::ManglerId);
    }

    let belop_text = deler.next().unwrap_or_default();
    if belop_text.trim().is_empty() {
        return Err(VedtakFeil::ManglerBelop);
    }

    let belop = belop_text.parse().map_err(|_| VedtakFeil::UgyldigBelop(belop_text.to_string()))?;

    Ok(Vedtak { id, belop })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(les("v1;250"), Ok(Vedtak { id: "v1".into(), belop: 250 }));
    }

    #[test]
    fn leser_mangler_id() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn leser_mangler_belop() {
        assert_eq!(les("v1;"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn leser_ugyldig_belop() {
        assert!(matches!(
            les("v1;abc"),
            Err(VedtakFeil::UgyldigBelop(ref verdi)) if verdi == "abc"
        ));
        assert!(les("v1;abc").unwrap_err().to_string().contains("abc"));
    }
}
