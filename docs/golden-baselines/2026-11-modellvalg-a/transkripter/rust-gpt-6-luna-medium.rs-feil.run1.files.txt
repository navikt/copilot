/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, PartialEq, Eq, thiserror::Error)]
pub enum VedtakFeil {
    #[error("Vedtak mangler id")]
    ManglerId,
    #[error("Vedtak mangler beløp")]
    ManglerBelop,
    #[error("Ugyldig beløp: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler
        .next()
        .filter(|id| !id.is_empty())
        .ok_or(VedtakFeil::ManglerId)?
        .to_string();
    let belop = deler
        .next()
        .filter(|belop| !belop.is_empty())
        .ok_or(VedtakFeil::ManglerBelop)?;
    let belop = belop
        .parse()
        .map_err(|_| VedtakFeil::UgyldigBelop(belop.to_string()))?;
    Ok(Vedtak { id, belop })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(
            les("v1;250"),
            Ok(Vedtak {
                id: "v1".into(),
                belop: 250
            })
        );
    }

    #[test]
    fn returnerer_feil_nar_id_mangler() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn returnerer_feil_nar_belop_mangler() {
        assert_eq!(les("v1;"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn returnerer_feil_med_ugyldig_belopstekst() {
        let feil = les("v1;ukjent").unwrap_err();

        assert_eq!(feil, VedtakFeil::UgyldigBelop("ukjent".into()));
        assert_eq!(feil.to_string(), "Ugyldig beløp: ukjent");
    }
}
