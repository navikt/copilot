/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, PartialEq, thiserror::Error)]
pub enum VedtakFeil {
    #[error("Vedtak mangler id")]
    ManglerId,
    #[error("Vedtak mangler beløp")]
    ManglerBelop,
    #[error("Ugyldig beløp: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
///
/// # Errors
///
/// Returnerer `ManglerId` hvis ID-en er tom, `ManglerBelop` hvis beløpet mangler,
/// eller `UgyldigBelop` hvis beløpet ikke kan parses som et heltall.
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler.next().unwrap_or_default();
    if id.is_empty() {
        return Err(VedtakFeil::ManglerId);
    }

    let belop = deler.next().ok_or(VedtakFeil::ManglerBelop)?;
    let belop = belop
        .parse()
        .map_err(|_| VedtakFeil::UgyldigBelop(belop.to_string()))?;

    Ok(Vedtak {
        id: id.to_string(),
        belop,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(
            les("v1;250"),
            Ok(Vedtak {
                id: "v1".into(),
                belop: 250
            })
        );
    }

    #[test]
    fn avviser_tom_id() {
        assert_eq!(les(";250"), Err(VedtakFeil::ManglerId));
    }

    #[test]
    fn avviser_manglende_belop() {
        assert_eq!(les("v1"), Err(VedtakFeil::ManglerBelop));
    }

    #[test]
    fn avviser_ugyldig_belop_og_tar_med_teksten_i_feilmeldingen() {
        let feil = les("v1;ugyldig").unwrap_err();

        assert_eq!(feil, VedtakFeil::UgyldigBelop("ugyldig".into()));
        assert!(feil.to_string().contains("ugyldig"));
    }
}
