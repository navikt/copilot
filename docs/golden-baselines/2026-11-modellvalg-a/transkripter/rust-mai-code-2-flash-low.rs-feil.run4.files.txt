/// Et vedtak lest fra en linje i filen fra fagsystemet.
#[derive(Debug, PartialEq)]
pub struct Vedtak {
    pub id: String,
    pub belop: u32,
}

#[derive(Debug, PartialEq, thiserror::Error)]
pub enum VedtakFeil {
    #[error("Mangler id")]
    ManglerId,
    #[error("Mangler belop")]
    ManglerBelop,
    #[error("Ugyldig belop: {0}")]
    UgyldigBelop(String),
}

/// Leser en linje på formen "id;belop".
pub fn les(linje: &str) -> Result<Vedtak, VedtakFeil> {
    let mut deler = linje.split(';');
    let id = deler.next().unwrap_or("").to_string();
    if id.is_empty() {
        return Err(VedtakFeil::ManglerId);
    }

    let belop_tekst = deler.next().unwrap_or("");
    if belop_tekst.is_empty() {
        return Err(VedtakFeil::ManglerBelop);
    }

    let belop = belop_tekst
        .parse()
        .map_err(|_| VedtakFeil::UgyldigBelop(belop_tekst.to_string()))?;
    Ok(Vedtak { id, belop })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn leser_gyldig_linje() {
        assert_eq!(
            les("v1;250"),
            Ok(Vedtak { id: "v1".into(), belop: 250 })
        );
    }

    #[test]
    fn mangler_id() {
        let feil = les(";250").unwrap_err();
        assert_eq!(feil, VedtakFeil::ManglerId);
        assert_eq!(feil.to_string(), "Mangler id");
    }

    #[test]
    fn mangler_belop() {
        let feil = les("v1;").unwrap_err();
        assert_eq!(feil, VedtakFeil::ManglerBelop);
        assert_eq!(feil.to_string(), "Mangler belop");
    }

    #[test]
    fn ugyldig_belop() {
        let feil = les("v1;hundre").unwrap_err();
        assert_eq!(
            feil,
            VedtakFeil::UgyldigBelop("hundre".to_string())
        );
        assert_eq!(feil.to_string(), "Ugyldig belop: hundre");
    }
}
