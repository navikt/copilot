package no.nav.vedtak

import org.apache.kafka.clients.consumer.Consumer
import org.apache.kafka.common.TopicPartition
import java.time.Duration

fun interface Utbetaling {
    fun utbetal(eventId: String, belop: Int)
}

// Leser vedtak fra topicen og utbetaler. Verdien i hver melding er "eventId;belop".
class VedtakKonsument(
    private val consumer: Consumer<String, String>,
    private val utbetaling: Utbetaling,
) {
    private val utbetalteEventIder = mutableSetOf<String>()

    fun poll() {
        val records = consumer.poll(Duration.ofMillis(100)).toList()
        try {
            for (record in records) {
                val (eventId, belop) = record.value().split(";")
                if (eventId !in utbetalteEventIder) {
                    utbetaling.utbetal(eventId, belop.toInt())
                    utbetalteEventIder += eventId
                }
            }
        } catch (e: Exception) {
            records.groupBy { TopicPartition(it.topic(), it.partition()) }.forEach { (partition, partitionRecords) ->
                try {
                    consumer.seek(partition, partitionRecords.minOf { it.offset() })
                } catch (seekException: Exception) {
                    e.addSuppressed(seekException)
                }
            }
            throw e
        }
        consumer.commitSync()
    }
}

package no.nav.vedtak

import org.apache.kafka.clients.consumer.ConsumerRecord
import org.apache.kafka.clients.consumer.MockConsumer
import org.apache.kafka.clients.consumer.OffsetResetStrategy
import org.apache.kafka.common.TopicPartition
import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith
import kotlin.test.assertNull

class VedtakKonsumentTest {
    private val tp = TopicPartition("vedtak", 0)

    private fun consumer(vararg verdier: String) = MockConsumer<String, String>(OffsetResetStrategy.EARLIEST).apply {
        assign(listOf(tp))
        updateBeginningOffsets(mapOf(tp to 0L))
        verdier.forEachIndexed { i, v -> addRecord(ConsumerRecord("vedtak", 0, i.toLong(), "sak-1", v)) }
    }

    @Test
    fun `utbetaler hvert vedtak`() {
        val utbetalt = mutableListOf<String>()
        VedtakKonsument(consumer("e1;100", "e2;200")) { id, _ -> utbetalt += id }.poll()
        assertEquals(listOf("e1", "e2"), utbetalt)
    }

    @Test
    fun `utbetaler ikke samme eventId på nytt`() {
        val kafkaConsumer = consumer("e1;100")
        val utbetalt = mutableListOf<String>()
        val vedtakKonsument = VedtakKonsument(kafkaConsumer) { id, _ -> utbetalt += id }

        vedtakKonsument.poll()
        kafkaConsumer.seek(tp, 0L)
        kafkaConsumer.addRecord(ConsumerRecord("vedtak", 0, 0L, "sak-1", "e1;100"))
        vedtakKonsument.poll()

        assertEquals(listOf("e1"), utbetalt)
        assertEquals(1L, kafkaConsumer.committed(setOf(tp))[tp]?.offset())
    }

    @Test
    fun `vedtaket blir forsøkt på nytt hvis utbetalingen feiler`() {
        val kafkaConsumer = consumer("e1;100", "e2;200")
        val utbetalt = mutableListOf<String>()
        var feile = true
        val vedtakKonsument = VedtakKonsument(kafkaConsumer) { id, _ ->
            if (id == "e2" && feile) {
                feile = false
                error("Utbetaling feilet")
            }
            utbetalt += id
        }

        assertFailsWith<IllegalStateException> { vedtakKonsument.poll() }

        assertNull(kafkaConsumer.committed(setOf(tp))[tp])
        assertEquals(0L, kafkaConsumer.position(tp))

        kafkaConsumer.addRecord(ConsumerRecord("vedtak", 0, 0L, "sak-1", "e1;100"))
        kafkaConsumer.addRecord(ConsumerRecord("vedtak", 0, 1L, "sak-1", "e2;200"))
        vedtakKonsument.poll()

        assertEquals(listOf("e1", "e2"), utbetalt)
        assertEquals(2L, kafkaConsumer.committed(setOf(tp))[tp]?.offset())
    }
}
