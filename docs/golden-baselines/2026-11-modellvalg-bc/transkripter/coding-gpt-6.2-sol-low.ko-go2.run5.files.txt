package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker.
func Rapport(navn string, grunnlag, grad int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad))
}

=== ./ytelse/utbetaling.go
package ytelse

// Utbetaling gir dagsatsen etter gradering, rundet ned til hele kroner.
// Grad er prosent av full ytelse (0 til 100).
func Utbetaling(grunnlag, grad int) int {
	const divisor = 260 * 100
	return (grunnlag/divisor)*grad + (grunnlag%divisor)*grad/divisor
}
