import { useState } from "react";
import { Button, Tag } from "@navikt/ds-react";
import { TrashIcon } from "@navikt/aksel-icons";

interface StatusPanelProps {
  status: string;
  onSlett: () => void;
}

export function StatusPanel({ status, onSlett }: StatusPanelProps) {
  const [apen, setApen] = useState(false);
  return (
    <div className="p-4 mx-8">
      <Button
        variant="tertiary"
        aria-expanded={apen}
        onClick={() => setApen(!apen)}
      >
        Vis detaljer
      </Button>
      <span role="status">
        <Tag variant={status === "feil" ? "error" : "success"}>{status}</Tag>
      </span>
      <Button
        variant="tertiary"
        icon={<TrashIcon title="Slett" />}
        onClick={onSlett}
      />
    </div>
  );
}
