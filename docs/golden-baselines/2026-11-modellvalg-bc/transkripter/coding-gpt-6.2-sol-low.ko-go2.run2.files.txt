package ytelse

import "fmt"

// Rapport beskriver dagsatsen til en bruker.
func Rapport(navn string, grunnlag, grad int) string {
	return fmt.Sprintf("%s får %d kr per dag", navn, Utbetaling(grunnlag, grad))
}

package ytelse

// Utbetaling gir dagsatsen: grunnlaget delt på 260 virkedager, rundet ned.

	// TODO: Bruk graden til å beregne den graderte dagsatsen.
