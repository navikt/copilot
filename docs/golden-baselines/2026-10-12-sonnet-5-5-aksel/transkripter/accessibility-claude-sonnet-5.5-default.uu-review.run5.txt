```tsx
import { useState } from "react";
import { TrashIcon } from "@navikt/aksel-icons";
import { Box, Button, HStack, Tag } from "@navikt/ds-react";

type Status = "ok" | "feil";

export function StatusPanel({ status, onSlett }: { status: Status; onSlett: () => void }) {
  const [apen, setApen] = useState(false);
  return (
    <Box padding="space-16">
      <HStack gap="space-8" align="center">
        <Button variant="tertiary" aria-expanded={apen} aria-controls="status-detaljer"
                onClick={() => setApen(!apen)}>
          Vis detaljer
        </Button>
        <span role="status">
          <Tag variant={status === "feil" ? "error" : "success"}>{status}</Tag>
        </span>
        <Button variant="tertiary" icon={<TrashIcon title="Slett" />} onClick={onSlett} />
      </HStack>
      {apen && <div id="status-detaljer">{/* detaljer */}</div>}
    </Box>
  );
}
```
