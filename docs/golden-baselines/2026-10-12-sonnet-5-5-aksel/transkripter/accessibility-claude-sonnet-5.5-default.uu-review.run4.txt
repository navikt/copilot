```tsx
import { useState } from "react";
import { TrashIcon } from "@navikt/aksel-icons";
import { Box, Button, HStack, Tag, VStack } from "@navikt/ds-react";

type Status = "feil" | "ok";
interface Props {
  status: Status;
  onSlett: () => void;
}

export function StatusPanel({ status, onSlett }: Props) {
  const [apen, setApen] = useState(false);
  return (
    <Box padding="space-16" marginInline="space-32">
      <VStack gap="space-8">
        <HStack gap="space-8" align="center">
          <Button
            variant="tertiary"
            aria-expanded={apen}
            aria-controls="status-detaljer"
            onClick={() => setApen(!apen)}
          >
            Vis detaljer
          </Button>
          <Tag variant={status === "feil" ? "error" : "success"} role="status">
            {status === "feil" ? "Feil" : "OK"}
          </Tag>
          <Button
            variant="danger"
            icon={<TrashIcon aria-hidden />}
            onClick={onSlett}
          >
            Slett
          </Button>
        </HStack>
        <div id="status-detaljer" hidden={!apen}>
          {/* detaljer */}
        </div>
      </VStack>
    </Box>
  );
}
```

Endringene:
- `Button` gir tastatur, fokusring og riktig rolle.
- `Tag` gir kontrast og tekst i tillegg til farge.
- Slettknappen har synlig tekst. Ønsker du bare ikon, bruk `title="Slett"` på `TrashIcon` og `aria-label`.
- Spacing-tokenene (`space-16`, `space-32`) forutsetter Aksel v8+. Sjekk versjonen din.

Bekreftelsesdialog (punkt 7) krever et valg: `Modal` i komponenten, eller bekreftelse hos kallende kode. Det er ikke lagt inn i forslaget.
