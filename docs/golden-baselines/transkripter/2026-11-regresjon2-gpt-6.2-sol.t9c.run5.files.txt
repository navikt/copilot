package no.nav.demo

import io.kotest.assertions.throwables.shouldThrow
import io.kotest.core.spec.style.FunSpec
import io.kotest.matchers.shouldBe

class FeriepengerTest : FunSpec({
    test("bruker 10,2 prosent før året personen fyller 60") {
        feriepenger(1_000L, 59) shouldBe 102L
    }

    test("bruker 12,5 prosent fra og med året personen fyller 60") {
        feriepenger(1_000L, 60) shouldBe 125L
        feriepenger(1_000L, 61) shouldBe 125L
    }

    test("runder ned under 50 øre og opp fra og med 50 øre") {
        feriepenger(24L, 59) shouldBe 2L
        feriepenger(25L, 59) shouldBe 3L
        feriepenger(26L, 59) shouldBe 3L
    }

    test("gir null kroner når grunnlaget er null") {
        feriepenger(0L, 60) shouldBe 0L
    }

    test("avviser negativt grunnlag") {
        shouldThrow<IllegalArgumentException> { feriepenger(-1L, 59) }
    }
})
