=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Grunnlaget kan ikke være negativt" }

    val sats = if (alderIOpptjeningsåret >= 60) 125 else 102
    return (grunnlag * sats + 500) / 1000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `gir ti og to prosent før året personen fyller seksti`() {
        assertEquals(102, feriepenger(1000, 59))
    }

    @Test
    fun `gir tolv og en halv prosent fra og med året personen fyller seksti`() {
        assertEquals(125, feriepenger(1000, 60))
        assertEquals(125, feriepenger(1000, 61))
    }

    @Test
    fun `runder opp ved nøyaktig femti øre`() {
        assertEquals(51, feriepenger(500, 59))
        assertEquals(63, feriepenger(500, 60))
    }

    @Test
    fun `runder til nærmeste hele krone`() {
        assertEquals(50, feriepenger(490, 59))
        assertEquals(52, feriepenger(510, 59))
    }

    @Test
    fun `gir null for null grunnlag`() {
        assertEquals(0, feriepenger(0, 59))
    }

    @Test
    fun `avviser negativt grunnlag`() {
        assertFailsWith<IllegalArgumentException> {
            feriepenger(-1, 59)
        }
    }
}
