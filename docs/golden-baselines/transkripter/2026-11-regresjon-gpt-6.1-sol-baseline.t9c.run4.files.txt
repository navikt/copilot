=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `uses 10 point 2 percent below age 60`() {
        assertEquals(10_200L, feriepenger(100_000L, 59))
    }

    @Test
    fun `uses 12 point 5 percent from age 60`() {
        assertEquals(12_500L, feriepenger(100_000L, 60))
        assertEquals(12_500L, feriepenger(100_000L, 61))
    }

    @Test
    fun `zero basis gives zero for both rates`() {
        assertEquals(0L, feriepenger(0L, 59))
        assertEquals(0L, feriepenger(0L, 60))
    }

    @Test
    fun `rounds below half down and half or above up at standard rate`() {
        assertEquals(25L, feriepenger(249L, 59))
        assertEquals(26L, feriepenger(250L, 59))
        assertEquals(26L, feriepenger(251L, 59))
    }

    @Test
    fun `rounds below half down and half or above up at higher rate`() {
        assertEquals(0L, feriepenger(3L, 60))
        assertEquals(1L, feriepenger(4L, 60))
        assertEquals(1L, feriepenger(5L, 60))
    }

    @Test
    fun `rejects negative basis for both rates`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 59) }
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 60) }
    }
}
