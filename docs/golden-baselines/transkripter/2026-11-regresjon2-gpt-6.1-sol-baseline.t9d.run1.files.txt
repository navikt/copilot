=== ./src/test/kotlin/no/nav/demo/AvrundingTest.kt
package no.nav.demo

import java.math.BigDecimal
import org.junit.jupiter.api.Assertions.assertEquals
import org.junit.jupiter.api.Test

class AvrundingTest {
    @Test
    fun `runder ned under femti ore`() {
        assertEquals(10L, avrundTilKrone(BigDecimal("10.49")))
    }

    @Test
    fun `runder femti ore og mer opp`() {
        assertEquals(11L, avrundTilKrone(BigDecimal("10.50")))
        assertEquals(11L, avrundTilKrone(BigDecimal("10.51")))
    }

    @Test
    fun `beholder null og hele kroner`() {
        assertEquals(0L, avrundTilKrone(BigDecimal.ZERO))
        assertEquals(10L, avrundTilKrone(BigDecimal("10.00")))
    }

    @Test
    fun `beholder presisjon for store belop`() {
        assertEquals(9007199254740993L, avrundTilKrone(BigDecimal("9007199254740992.50")))
    }
}

package no.nav.demo

import org.junit.jupiter.api.Assertions.assertEquals
import org.junit.jupiter.api.Assertions.assertThrows
import org.junit.jupiter.api.Test

class FeriepengerTest {
    @Test
    fun `gir 10 komma 2 prosent til og med 59 ar`() {
        assertEquals(10200L, feriepenger(100000L, 30))
        assertEquals(10200L, feriepenger(100000L, 59))
    }

    @Test
    fun `gir 12 komma 5 prosent fra og med 60 ar`() {
        assertEquals(12500L, feriepenger(100000L, 60))
        assertEquals(12500L, feriepenger(100000L, 61))
    }

    @Test
    fun `runder til naermeste krone med femti ore opp`() {
        assertEquals(2L, feriepenger(24L, 59))
        assertEquals(26L, feriepenger(250L, 59))
        assertEquals(2L, feriepenger(19L, 60))
        assertEquals(3L, feriepenger(20L, 60))
        assertEquals(3L, feriepenger(21L, 60))
    }

    @Test
    fun `gir null for null grunnlag`() {
        assertEquals(0L, feriepenger(0L, 59))
        assertEquals(0L, feriepenger(0L, 60))
    }

    @Test
    fun `avviser negativt grunnlag uansett sats`() {
        assertThrows(IllegalArgumentException::class.java) { feriepenger(-1L, 59) }
        assertThrows(IllegalArgumentException::class.java) { feriepenger(-1L, 60) }
    }
}
