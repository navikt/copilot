=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Grunnlaget kan ikke være negativt" }

    val prosent = if (alderIOpptjeningsåret >= 60) 125 else 102
    val antall = grunnlag * prosent

    return (antall + 500) / 1000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `gir ti komma to prosent for personer under seksti år`() {
        assertEquals(102, feriepenger(grunnlag = 1000, alderIOpptjeningsåret = 59))
    }

    @Test
    fun `gir tolv komma fem prosent fra og med seksti år`() {
        assertEquals(125, feriepenger(grunnlag = 1000, alderIOpptjeningsåret = 60))
    }

    @Test
    fun `runder femti øre opp til nærmeste krone`() {
        assertEquals(1, feriepenger(grunnlag = 10, alderIOpptjeningsåret = 59))
    }

    @Test
    fun `runder ned under femti øre`() {
        assertEquals(0, feriepenger(grunnlag = 4, alderIOpptjeningsåret = 59))
    }

    @Test
    fun `gir null for null grunnlag`() {
        assertEquals(0, feriepenger(grunnlag = 0, alderIOpptjeningsåret = 60))
    }

    @Test
    fun `kaster IllegalArgumentException for negativt grunnlag`() {
        assertFailsWith<IllegalArgumentException> {
            feriepenger(grunnlag = -1, alderIOpptjeningsåret = 59)
        }
    }
}
