=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `satsen øker fra og med året personen fyller 60`() {
        assertEquals(102L, feriepenger(1_000L, 59))
        assertEquals(125L, feriepenger(1_000L, 60))
        assertEquals(125L, feriepenger(1_000L, 61))
    }

    @Test
    fun `under 50 øre rundes ned og 50 øre rundes opp`() {
        assertEquals(12L, feriepenger(122L, 59))
        assertEquals(13L, feriepenger(100L, 60))
        assertEquals(26L, feriepenger(251L, 59))
    }

    @Test
    fun `null kroner gir null feriepenger`() {
        assertEquals(0L, feriepenger(0L, 60))
    }

    @Test
    fun `negativt grunnlag avvises`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 59) }
    }
}
