package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "grunnlag must be non-negative" }

    val promille = if (alderIOpptjeningsåret >= 60) 125L else 102L
    return (grunnlag * promille + 500) / 1000
}

package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `uses ordinary rate until age 59 and higher rate from age 60`() {
        assertEquals(10_200L, feriepenger(100_000, 59))
        assertEquals(12_500L, feriepenger(100_000, 60))
        assertEquals(12_500L, feriepenger(100_000, 61))
    }

    @Test
    fun `rounds ordinary rate to nearest krone with half rounded up`() {
        assertEquals(25L, feriepenger(249, 59))
        assertEquals(26L, feriepenger(250, 59))
        assertEquals(26L, feriepenger(251, 59))
    }

    @Test
    fun `rounds higher rate to nearest krone with half rounded up`() {
        assertEquals(0L, feriepenger(3, 60))
        assertEquals(1L, feriepenger(4, 60))
        assertEquals(1L, feriepenger(5, 60))
    }

    @Test
    fun `zero basis gives zero for both rates`() {
        assertEquals(0L, feriepenger(0, 59))
        assertEquals(0L, feriepenger(0, 60))
    }

    @Test
    fun `rejects negative basis for both rates`() {
        for (age in listOf(59, 60)) {
            assertFailsWith<IllegalArgumentException> {
                feriepenger(-1, age)
            }
        }
    }
}
