package no.nav.demo

fun avrundTilNaermesteKrone(belopIOre: Long): Long {
    require(belopIOre >= 0) { "Beløpet kan ikke være negativt" }
    val heleKroner = belopIOre / 100
    val resterendeORe = belopIOre % 100

    return if (resterendeORe >= 50) heleKroner + 1 else heleKroner
}

=== ./src/test/kotlin/no/nav/demo/AvrundingTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class AvrundingTest {
    @Test
    fun `avrunder under 50 ore ned`() {
        assertEquals(123, avrundTilNaermesteKrone(12_349))
    }

    @Test
    fun `avrunder 50 ore opp`() {
        assertEquals(124, avrundTilNaermesteKrone(12_350))
    }

    @Test
    fun `avrunder over 50 ore opp`() {
        assertEquals(124, avrundTilNaermesteKrone(12_351))
    }

    @Test
    fun `avviser negativt belop`() {
        assertFailsWith<IllegalArgumentException> {
            avrundTilNaermesteKrone(-1)
        }
    }
}
