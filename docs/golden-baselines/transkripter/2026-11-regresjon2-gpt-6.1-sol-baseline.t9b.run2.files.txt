=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Grunnlaget kan ikke være negativt" }

    val promille = if (alderIOpptjeningsåret >= 60) 125L else 102L
    return (grunnlag * promille + 500) / 1000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `gir 10 komma 2 prosent til og med 59 år`() {
        assertEquals(10_200L, feriepenger(100_000, 59))
    }

    @Test
    fun `gir 12 komma 5 prosent fra og med 60 år`() {
        assertEquals(12_500L, feriepenger(100_000, 60))
        assertEquals(12_500L, feriepenger(100_000, 61))
    }

    @Test
    fun `null i grunnlag gir null for begge satser`() {
        assertEquals(0L, feriepenger(0, 59))
        assertEquals(0L, feriepenger(0, 60))
    }

    @Test
    fun `avrunder ned under 50 øre og opp fra 50 øre med ordinær sats`() {
        assertEquals(2L, feriepenger(24, 59))
        assertEquals(26L, feriepenger(250, 59))
        assertEquals(3L, feriepenger(25, 59))
    }

    @Test
    fun `avrunder ned under 50 øre og opp fra 50 øre med høy sats`() {
        assertEquals(0L, feriepenger(3, 60))
        assertEquals(1L, feriepenger(4, 60))
        assertEquals(1L, feriepenger(5, 60))
    }

    @Test
    fun `avviser negativt grunnlag for begge satser`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1, 59) }
        assertFailsWith<IllegalArgumentException> { feriepenger(-1, 60) }
    }
}
