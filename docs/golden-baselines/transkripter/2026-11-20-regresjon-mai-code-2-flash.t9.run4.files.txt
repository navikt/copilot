=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Grunnlag kan ikke være negativt" }

    val prosentPromille = if (alderIOpptjeningsåret >= 60) 125 else 102
    return (grunnlag * prosentPromille + 500) / 1000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import org.junit.jupiter.api.Assertions.assertEquals
import org.junit.jupiter.api.Assertions.assertThrows
import org.junit.jupiter.api.Test

class FeriepengerTest {
    @Test
    fun `gir 10,2 prosent under 60 ar`() {
        assertEquals(102, feriepenger(1000, 59))
    }

    @Test
    fun `gir 12,5 prosent fra 60 ar`() {
        assertEquals(125, feriepenger(1000, 60))
    }

    @Test
    fun `runder 50 ore opp til naermeste krone`() {
        assertEquals(51, feriepenger(500, 59))
        assertEquals(63, feriepenger(500, 60))
    }

    @Test
    fun `gir null for null grunnlag`() {
        assertEquals(0, feriepenger(0, 59))
    }

    @Test
    fun `kaster IllegalArgumentException for negativt grunnlag`() {
        assertThrows(IllegalArgumentException::class.java) {
            feriepenger(-1, 59)
        }
    }
}
