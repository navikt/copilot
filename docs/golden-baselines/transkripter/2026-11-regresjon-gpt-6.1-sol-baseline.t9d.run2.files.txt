package no.nav.demo

import java.math.BigDecimal
import kotlin.test.Test
import kotlin.test.assertEquals

class AvrundingTest {
    @Test
    fun `rounds down below fifty ore`() {
        assertEquals(10L, avrundTilHeleKroner(BigDecimal("10.49")))
    }

    @Test
    fun `rounds up from fifty ore`() {
        assertEquals(11L, avrundTilHeleKroner(BigDecimal("10.50")))
        assertEquals(11L, avrundTilHeleKroner(BigDecimal("10.51")))
    }

    @Test
    fun `preserves whole kroner and zero`() {
        assertEquals(10L, avrundTilHeleKroner(BigDecimal("10.00")))
        assertEquals(0L, avrundTilHeleKroner(BigDecimal.ZERO))
    }
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `uses standard rate before sixty`() {
        assertEquals(10_200L, feriepenger(100_000L, 59))
    }

    @Test
    fun `uses higher rate from the year of turning sixty`() {
        assertEquals(12_500L, feriepenger(100_000L, 60))
        assertEquals(12_500L, feriepenger(100_000L, 61))
    }

    @Test
    fun `rounds to nearest krone with halves up`() {
        assertEquals(2L, feriepenger(24L, 59))
        assertEquals(3L, feriepenger(25L, 59))
        assertEquals(0L, feriepenger(3L, 60))
        assertEquals(1L, feriepenger(4L, 60))
    }

    @Test
    fun `zero basis gives zero`() {
        assertEquals(0L, feriepenger(0L, 59))
        assertEquals(0L, feriepenger(0L, 60))
    }

    @Test
    fun `rejects negative basis`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 59) }
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 60) }
    }
}
