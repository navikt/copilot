=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Feriepengegrunnlaget kan ikke være negativt" }

    val satsIPromille = if (alderIOpptjeningsåret >= 60) 125L else 102L
    return (grunnlag * satsIPromille + 500) / 1_000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `bruker hoyere sats fra og med 60 aar`() {
        assertEquals(102L, feriepenger(1_000, 59))
        assertEquals(125L, feriepenger(1_000, 60))
        assertEquals(125L, feriepenger(1_000, 61))
    }

    @Test
    fun `runder til naermeste krone med halvkroner opp`() {
        assertEquals(50L, feriepenger(495, 59))
        assertEquals(26L, feriepenger(250, 59))
        assertEquals(26L, feriepenger(251, 59))
        assertEquals(1L, feriepenger(4, 60))
    }

    @Test
    fun `gir null ved null grunnlag`() {
        assertEquals(0L, feriepenger(0, 60))
    }

    @Test
    fun `avviser negativt grunnlag`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1, 59) }
    }
}
