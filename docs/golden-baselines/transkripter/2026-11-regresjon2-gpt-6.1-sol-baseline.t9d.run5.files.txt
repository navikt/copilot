=== ./src/test/kotlin/no/nav/demo/AvrundingTest.kt
package no.nav.demo

import java.math.BigDecimal
import kotlin.test.Test
import kotlin.test.assertEquals

class AvrundingTest {
    @Test
    fun `rounds down below fifty ore`() {
        assertEquals(10L, avrundTilKrone(BigDecimal("10.49")))
    }

    @Test
    fun `rounds fifty ore up`() {
        assertEquals(11L, avrundTilKrone(BigDecimal("10.50")))
    }

    @Test
    fun `rounds up above fifty ore`() {
        assertEquals(11L, avrundTilKrone(BigDecimal("10.51")))
    }

    @Test
    fun `preserves whole kroner and zero`() {
        assertEquals(10L, avrundTilKrone(BigDecimal("10.00")))
        assertEquals(0L, avrundTilKrone(BigDecimal.ZERO))
    }
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `uses standard rate below sixty`() {
        assertEquals(10_200L, feriepenger(100_000L, 59))
    }

    @Test
    fun `uses higher rate from sixty onwards`() {
        assertEquals(12_500L, feriepenger(100_000L, 60))
        assertEquals(12_500L, feriepenger(100_000L, 61))
    }

    @Test
    fun `rounds standard rate to nearest krone`() {
        assertEquals(25L, feriepenger(249L, 59))
        assertEquals(26L, feriepenger(250L, 59))
        assertEquals(26L, feriepenger(251L, 59))
    }

    @Test
    fun `rounds higher rate to nearest krone`() {
        assertEquals(12L, feriepenger(99L, 60))
        assertEquals(13L, feriepenger(100L, 60))
        assertEquals(13L, feriepenger(101L, 60))
    }

    @Test
    fun `zero basis gives zero`() {
        assertEquals(0L, feriepenger(0L, 59))
        assertEquals(0L, feriepenger(0L, 60))
    }

    @Test
    fun `rejects negative basis`() {
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 59) }
        assertFailsWith<IllegalArgumentException> { feriepenger(-1L, 60) }
    }
}
