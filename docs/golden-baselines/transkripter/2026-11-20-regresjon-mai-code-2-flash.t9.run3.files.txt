=== ./src/main/kotlin/no/nav/demo/Feriepenger.kt
package no.nav.demo

fun feriepenger(grunnlag: Long, alderIOpptjeningsåret: Int): Long {
    require(grunnlag >= 0) { "Grunnlaget kan ikke være negativt" }

    val satsPromille = if (alderIOpptjeningsåret >= 60) 125 else 102
    return (grunnlag * satsPromille + 500) / 1000
}

=== ./src/test/kotlin/no/nav/demo/FeriepengerTest.kt
package no.nav.demo

import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class FeriepengerTest {
    @Test
    fun `gir 10,2 prosent under 60 ar`() {
        assertEquals(102, feriepenger(1000, 59))
    }

    @Test
    fun `gir 12,5 prosent fra og med 60 ar`() {
        assertEquals(125, feriepenger(1000, 60))
    }

    @Test
    fun `runder 50 ore opp til narmeste krone`() {
        assertEquals(26, feriepenger(250, 59))
        assertEquals(3, feriepenger(4, 60))
    }

    @Test
    fun `gir 0 for grunnlag 0`() {
        assertEquals(0, feriepenger(0, 59))
        assertEquals(0, feriepenger(0, 60))
    }

    @Test
    fun `kaster IllegalArgumentException for negativt grunnlag`() {
        assertFailsWith<IllegalArgumentException> {
            feriepenger(-1, 59)
        }
    }
}
